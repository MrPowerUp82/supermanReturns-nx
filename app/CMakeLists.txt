cmake_minimum_required(VERSION 3.25)
project(superman_returns VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/generated/default/sources.cmake")
    message(FATAL_ERROR "Run python tools/project.py codegen --rexglue <host rexglue executable> first")
endif()
set(SR_CODEGEN_STAMP "${CMAKE_CURRENT_SOURCE_DIR}/generated/default/manifest.sha256")
if(NOT EXISTS "${SR_CODEGEN_STAMP}")
    message(FATAL_ERROR "Codegen stamp missing. Generate with tools/project.py codegen.")
endif()
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/superman_returns_manifest.toml" SR_MANIFEST_HASH)
file(READ "${SR_CODEGEN_STAMP}" SR_CODEGEN_HASH)
string(STRIP "${SR_CODEGEN_HASH}" SR_CODEGEN_HASH)
if(NOT SR_MANIFEST_HASH STREQUAL SR_CODEGEN_HASH)
    message(FATAL_ERROR "Manifest changed since codegen. Run tools/project.py codegen again.")
endif()
include(cmake/rexglue.cmake)

set(SR_SOURCES src/main.cpp src/sr_settings.cpp src/xma_fixes.cpp src/skip_intro.cpp
    src/sr_shader_library.cpp src/sr_shader_registry.cpp src/sr_shader_hooks.cpp
    src/sr_native_ring.cpp src/sr_native_shader_safety.cpp src/sr_native_system.cpp)
if(WIN32)
    add_executable(superman_returns WIN32 ${SR_SOURCES})
else()
    add_executable(superman_returns ${SR_SOURCES})
endif()
rexglue_setup_target(superman_returns GPU_PLUGINS xenos)

if(REXGLUE_PLATFORM_SWITCH)
    option(SR_LTO "Enable LTO for the application and recompiled game" OFF)
    if(SR_LTO)
        foreach(target superman_returns superman_returns_recomp)
            target_compile_options(${target} PRIVATE -flto -fno-fat-lto-objects)
        endforeach()
        target_link_options(superman_returns PRIVATE -flto=2 -flto-partition=balanced)
    endif()
    rexglue_switch_add_nro(superman_returns
        NAME "Superman Returns NX (experimental)"
        AUTHOR "Superman Returns NX contributors" VERSION "${PROJECT_VERSION}")
elseif(TARGET imgui)
    target_include_directories(superman_returns PRIVATE
        $<TARGET_PROPERTY:imgui,INTERFACE_INCLUDE_DIRECTORIES>)
endif()
